\begin{frame}{Task Definition}
\begin{block}{Goal}
Build a reproducible binary ECG screening pipeline from PTB-XL raw WFDB waveforms and metadata, then explain how fitted models behave in the high-dimensional feature regime.
\end{block}
\begin{itemize}
\item Input: ten-second, twelve-lead ECG waveform records plus patient and acquisition metadata.
\item Target: normal versus diagnostic abnormality from SCP diagnostic superclasses.
\item Data split: official PTB-XL folds; train, validation, and held-out test are kept separate.
\item Output: trained model comparison, final metrics, paper, presentation, and MLflow logs.
\end{itemize}
\end{frame}

\begin{frame}{Regularization Choices: Linear Models}
\begin{columns}[T,totalwidth=\textwidth]
\begin{column}{0.46\textwidth}
\begin{block}{Regularized objective}
\small
\[
\min_{\beta,b}
\frac{1}{n}\sum_i \ell(y_i,\beta^\top x_i+b)
+\lambda_1\|\beta\|_1+\frac{\lambda_2}{2}\|\beta\|_2^2 .
\]
\begin{itemize}
\item \(L_1\): sparsity and feature selection.
\item \(L_2\): shrinkage for correlated predictors.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.50\textwidth}
\begin{block}{Selected by Optuna}
\small
\begin{itemize}
\item Logistic regression: \(L_2\), \(C=0.164\). This applies nontrivial ridge shrinkage to multicollinear ECG feature groups.
\item SGD linear: \(L_2\), \(\alpha=2.07{\times}10^{-6}\). Elastic-net was searched but not selected.
\item Interpretation: the linear models kept shrinkage instead of strong sparsity because many weak waveform features can contribute jointly.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Random Matrix Theory Diagnostic}
\begin{block}{Marchenko--Pastur reference}
For standardized white-noise features with aspect ratio \(\gamma=p/n\), sample-covariance eigenvalues concentrate on
\[
\lambda_{\pm}=(1\pm\sqrt{\gamma})^2.
\]
\end{block}
\begin{itemize}
\item Eigenvalues inside \([\lambda_-,\lambda_+]\) form the empirical bulk.
\item Eigenvalues above \(\lambda_+\) are structured outliers.
\item The experiment checks whether model scores survive if the data are projected onto outlier or bulk subspaces.
\end{itemize}
\end{frame}

\begin{frame}{PCA Eigenvalue Problem}
\begin{block}{What PCA solves}
For the standardized design matrix \(X\in\mathbb{R}^{n\times p}\),
\[
S=\frac{1}{n-1}X^\top X = V\Lambda V^\top,
\qquad
z_{ij}=x_i^\top v_j.
\]
PCA orders directions \(v_j\) by sample eigenvalues \(\lambda_j\).
\end{block}
\begin{itemize}
\item In low dimensions, large sample variance is often a useful signal heuristic.
\item In high dimensions, even pure noise has a spread-out sample spectrum.
\item The question becomes: which PCA directions are outside what noise alone would produce?
\end{itemize}
\end{frame}

\begin{frame}{Marchenko--Pastur as a PCA Cutoff}
\begin{columns}[T,totalwidth=\textwidth]
\begin{column}{0.52\textwidth}
\begin{block}{Noise reference}
If standardized features were independent noise and \(p/n\to\gamma\), the noise eigenvalues concentrate in
\[
\lambda_{\pm}=(1\pm\sqrt{\gamma})^2.
\]
\end{block}
\end{column}
\begin{column}{0.44\textwidth}
\begin{block}{Observed feature map}
\begin{itemize}
\item \(n=17{,}084\), \(p=594\).
\item \(p/n=0.0348\).
\item MP support: \([0.662,1.408]\).
\item 60 eigenvalues exceed \(\lambda_+\).
\end{itemize}
\end{block}
\end{column}
\end{columns}
\vspace{0.15cm}
\small Components above \(\lambda_+\) are treated as candidate structured directions; components inside the bulk are not automatically useless, only statistically compatible with noise under the reference model.
\end{frame}

\begin{frame}{Projection, Bias, and Variance}
\[
I_{\mathrm{out}}=\{j:\lambda_j>\lambda_+\},\qquad
P_{\mathrm{out}}=V_{I_{\mathrm{out}}}V_{I_{\mathrm{out}}}^{\top},\qquad
P_{\mathrm{bulk}}=I-P_{\mathrm{out}}.
\]
\[
X_{\mathrm{out}}=XP_{\mathrm{out}},\qquad X_{\mathrm{bulk}}=XP_{\mathrm{bulk}}.
\]
\begin{itemize}
\item Hard PCA/RMT filtering keeps \(X_{\mathrm{out}}\) and discards \(X_{\mathrm{bulk}}\).
\item This can lower estimator variance by removing unstable directions.
\item It can also add approximation bias if weak real signal lives inside the bulk.
\item In this run, original XGBoost test ROC-AUC was \(0.9097\), while MP-outlier reconstruction was \(0.8957\) and PCA++ scores were \(0.8938\).
\end{itemize}
\end{frame}

\footnotesize
\begin{itemize}
\item Full features are best on test.
\item MP reconstruction nearly preserves F1: \(0.8283\) versus \(0.8328\).

\textbf{Interpretation}

\footnotesize
\begin{itemize}
\item Filtering removes bulk components and lowers variance.
\item The bulk is not pure noise; weak predictive signal remains there.
\item RMT is most useful here as a diagnostic, not as the final predictor.
\end{itemize}

\begin{frame}{Interpretation}
\begin{itemize}
\item The covariance spectrum has \(p/n=0.0348\), MP support \([0.662,1.408]\), and 60 outlier components.
\item Those outliers explain 76.9\% of standardized variance, so the feature map has a compact dominant subspace.
\item MP-outlier reconstructions preserve much of the predictive signal.
\item MP-bulk reconstructions are weaker but not empty; hard filtering removes weak useful information.
\item This explains why RMT is useful as model analysis here, but not as a direct accuracy booster.
\end{itemize}
\end{frame}
